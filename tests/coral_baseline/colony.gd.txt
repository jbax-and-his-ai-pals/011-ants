	# Direct CLI runs also load the extension without requiring an editor import.

		GDExtensionManager.load_extension("res://native/ant_kernel.gdextension")
	if ClassDB.class_exists("AntKernel"):
		native_kernel = ClassDB.instantiate("AntKernel")
	var normal := SystemFont.new()
	normal.font_names = PackedStringArray(["Segoe UI", "Noto Sans", "Arial"])
	regular_font = normal
	var semibold := SystemFont.new()
	semibold.font_names = normal.font_names
	semibold.font_weight = 600

	_build_map_view()

	get_viewport().size_changed.connect(_layout_ui)
	_layout_ui()

func _process(delta: float) -> void:
	var sim_start := Time.get_ticks_usec()
	if not paused and match_result.is_empty():
		accumulator += minf(delta, 0.15) * speed
		while accumulator >= STEP:
			simulate(STEP)
			accumulator -= STEP
			if not match_result.is_empty():
				accumulator = 0
				break
	last_sim_ms = float(Time.get_ticks_usec() - sim_start) / 1000.0
	smoothed_frame_ms = lerpf(smoothed_frame_ms, delta * 1000, 0.08)
	if following and selected_ant >= 0 and health[selected_ant] > 0:
		underground = layers[selected_ant] >= 0
		if underground: inspected_colony = layers[selected_ant]
		_activate_camera_view()
		camera_center = positions[selected_ant]
		_sync_camera()
	visual_clock += delta
	if visual_clock >= 0.15:
		_refresh_map_images()
		mini_canvas.queue_redraw()
		visual_clock = 0
	ui_clock += delta
	if ui_clock > 0.15:
		_update_ui()
		ui_clock = 0
	world_canvas.queue_redraw()

func simulate(dt: float) -> void:
	_check_outcome()
	if not match_result.is_empty():
		return
	var profile_start := Time.get_ticks_usec() if profiling else 0
	elapsed += dt
	tick_number += 1
	scent_clock += dt
	allocation_clock += dt
	if scent_clock >= 0.15:
		if native_enabled and native_kernel:
			native_kernel.decay_scent(self)
		else:
			for slot in range(active_scent.size() - 1, -1, -1):
				var j := active_scent[slot]
				food_scent[j] *= 0.993
				home_scent[j] *= 0.99
				rival_food_scent[j] *= 0.993
				rival_home_scent[j] *= 0.99
				third_food_scent[j] *= 0.993
				third_home_scent[j] *= 0.99
				if maxf(maxf(food_scent[j], home_scent[j]), maxf(maxf(rival_food_scent[j], rival_home_scent[j]), maxf(third_food_scent[j], third_home_scent[j]))) < 0.008:
					food_scent[j] = 0
					home_scent[j] = 0
					rival_food_scent[j] = 0
					rival_home_scent[j] = 0
					third_food_scent[j] = 0
					third_home_scent[j] = 0
					scent_active[j] = 0
					active_scent[slot] = active_scent[active_scent.size() - 1]
					active_scent.resize(active_scent.size() - 1)
					if trail_image: trail_image.set_pixel(j % COLS, j / COLS, Color.TRANSPARENT)
		trails_dirty = true
		scent_clock = 0
	if allocation_clock >= 0.8:
		_allocate_workers()
		command_system.allocate()
		if rival_ai_enabled:
			rival_ai.update()
			if colony_count == 3: third_ai.update()
		if invasions_enabled:
			_update_defenses()
			for colony in colony_count:
				if elapsed >= next_raids[colony] and not (colony == 0 and command_system.has_raid()) and not (colony > 0 and rival_ai_enabled):
					launch_raid(colony, 8)
					next_raids[colony] = elapsed + 60
		allocation_clock = 0
	for colony in colony_count:
		nest_quiet[colony] = 0 if invaders_in(colony) > 0 else nest_quiet[colony] + dt
		queen_flashes[colony] = maxf(0, queen_flashes[colony] - dt)
	var encounter_start := Time.get_ticks_usec() if profiling else 0
	_prepare_encounters(dt)
	var movement_start := Time.get_ticks_usec() if profiling else 0
	var processed := PackedByteArray()
	if native_enabled and native_kernel:
		# Free surface foragers do not read other workers' movement. They retain
		# their original relative food/scent order; fighters use the shared snapshot.
		# Mission workers cannot change forager eligibility within this tick.
		processed = native_kernel.forage_batch(self, dt)
	else:
		_rebuild_food_index()
	# Alternate the leading colony for shared-food pickup as well.
	var movers := PackedInt32Array()
	if not processed.is_empty(): movers = native_kernel.mission_workers(self, processed)
	else:
		for phase in colony_count: movers.append_array(colony_workers[(tick_number + phase) % colony_count])
	for i in movers:
		if health[i] <= 0:
			continue
		if not processed.is_empty() and processed[i]:
			continue
		var was_recovering: bool = roles[i] == RECOVERING
		if roles[i] != FORAGER:
			_recover_worker(i, dt)
		# Finish the rest tick at home; forage from the next tick in both kernels.
		if was_recovering and roles[i] == FORAGER: continue
		if opponents[i] >= 0:
			_fight(i, dt)
		elif layers[i] == SURFACE:
			if roles[i] == FORAGER:
				_forage(i, dt)
			else:
				_surface_mission(i, dt)
		else:
			if roles[i] == DIGGER:
				_dig(i, dt)
			else:
				_tunnel_mission(i, dt)
	var resolution_start := Time.get_ticks_usec() if profiling else 0
	if native_enabled and native_kernel:
		var dead: PackedInt32Array = native_kernel.resolve_tick(self)
		for i in dead: _kill_ant(i)
	else:
		_resolve_attacks()
		for i in health.size():
			if health[i] > 0 and layers[i] == layer_snapshot[i]:
				walked[i] += positions[i].distance_to(movement_snapshot[i])
		# Resolve damage together so neither colony gains a turn-order advantage.
		for i in health.size():
			if health[i] > 0 and combat_damage[i] > 0:
				health[i] = maxf(0, health[i] - combat_damage[i])
				hit_flashes[i] = 0.16
				if health[i] <= 0:
					_kill_ant(i)
	for colony in colony_count:
		if queen_health[colony] > 0 and queen_damage[colony] > 0:
			queen_health[colony] = maxf(0, queen_health[colony] - queen_damage[colony])
			queen_flashes[colony] = 0.2
			if queen_health[colony] == 0:
				queen_death_causes[colony] = "invaders"
	for k in range(remains.size() - 1, -1, -1):
		remains[k].time -= dt
		if remains[k].time <= 0:
			remains.remove_at(k)
	if reproduction_enabled:
		_update_nurseries(dt)
	command_system.tick()
	_check_outcome()
	if profiling:
		profile_usec[0] += encounter_start - profile_start
		profile_usec[1] += movement_start - encounter_start
		profile_usec[2] += resolution_start - movement_start
		profile_usec[3] += Time.get_ticks_usec() - resolution_start
		profile_ticks += 1

func starting_total() -> int:
	return workers_per_colony * colony_count

func colony_color(colony: int) -> Color:
	return [ACCENT, RIVAL, AZURE][colony]

func enemy_for(colony: int) -> int:
	var best := -1
	var distance := INF
	for candidate in colony_count:
		if candidate == colony or queen_health[candidate] <= 0: continue
		var d := nest_position(colony).distance_squared_to(nest_position(candidate))
		if d < distance: best = candidate; distance = d
	return best if best >= 0 else (1 if colony == 0 else 0)

func nest_for(i: int) -> Vector2:
	return nest_position(colonies[i])

func nest_position(colony: int) -> Vector2:
	if colony == 2: return THIRD_HOME
	return HOME if (colony == 0) != swapped_sides else RIVAL_HOME

func living_count(colony: int) -> int:
	if native_enabled and native_kernel: return native_kernel.living_count(self, colony)
	var result := 0
	for i in health.size():
		if colonies[i] == colony and health[i] > 0:
			result += 1
	return result

func _prepare_encounters(dt: float) -> void:
	if native_enabled and native_kernel:
		native_kernel.prepare_encounters(self, dt)
	else:
		_prepare_encounters_reference(dt)

func _prepare_encounters_reference(dt: float) -> void:
	movement_snapshot = positions.duplicate()
	layer_snapshot = layers.duplicate()
	encounter_neighbors.clear()
	opponents.fill(-1)
	combat_damage.fill(0)
	last_attackers.fill(-1)
	queen_damage.fill(0)
	bucket_heads.resize(BUCKET_CELLS * colony_count * (colony_count + 1))
	bucket_heads.fill(-1)
	bucket_next.resize(health.size())
	# Reverse insertion makes each linked bucket iterate in original slot order.
	for i in range(health.size() - 1, -1, -1):
		if attack_cooldowns[i] > 0: attack_cooldowns[i] = maxf(0, attack_cooldowns[i] - dt)
		if hit_flashes[i] > 0: hit_flashes[i] = maxf(0, hit_flashes[i] - dt)
		if health[i] <= 0:
			continue
		var cell := Vector2i(positions[i] / BUCKET_SIZE)
		var bucket := cell.x + cell.y * BUCKET_COLS + ((layers[i] + 1) * colony_count + colonies[i]) * BUCKET_CELLS
		bucket_next[i] = bucket_heads[bucket]
		bucket_heads[bucket] = i
	if not combat_enabled:
		return
	# Query only adjacent spatial buckets, rather than every ant against every other.
	for i in health.size():
		if health[i] <= 0:
			continue
		var p := positions[i]
		var radius := 34.0 if doctrines[colonies[i]] == 2 else ENCOUNTER_RADIUS
		if doctrines[colonies[i]] == 1 and carrying[i]:
			# Steward carriers prioritize getting food home over chasing an enemy.
			radius = ATTACK_RANGE
		var nearest := radius * radius
		var cell := Vector2i(p / BUCKET_SIZE)
		for enemy_colony in colony_count:
			if enemy_colony == colonies[i]: continue
			var enemy_base := ((layers[i] + 1) * colony_count + enemy_colony) * BUCKET_CELLS
			var key := cell.x + cell.y * BUCKET_COLS + enemy_base
			if not encounter_neighbors.has(key):
				var nearby: Array[int] = []
				# Dense heads avoid hashing vectors or allocating per-bucket arrays.
				# This broad phase is shared; target choice remains per ant, every tick.
				for y in range(maxi(0, cell.y - 2), mini(BUCKET_ROWS, cell.y + 3)):
					for x in range(maxi(0, cell.x - 2), mini(BUCKET_COLS, cell.x + 3)):
						var other := bucket_heads[x + y * BUCKET_COLS + enemy_base]
						while other >= 0:
							nearby.append(other)
							other = bucket_next[other]
				encounter_neighbors[key] = nearby
			for other in encounter_neighbors[key]:
				var distance := p.distance_squared_to(positions[other])
				if roles[i] == RECOVERING and distance > ATTACK_RANGE * ATTACK_RANGE: continue
				if colonies[i] == 0 and layers[i] == SURFACE and distance > ATTACK_RANGE * ATTACK_RANGE:
					if command_system.is_avoided(p) or command_system.is_avoided(positions[other]): continue
				if roles[i] == WARDEN and distance > ATTACK_RANGE * ATTACK_RANGE and positions[other].distance_squared_to(worker_order_goals[i]) > 110 * 110: continue
				if distance < nearest and (layers[i] == SURFACE or _tunnel_line_clear(p, positions[other], layers[i])):
					nearest = distance
					opponents[i] = other

func _fight(i: int, dt: float) -> void:
	var enemy := opponents[i]
	if layers[i] != layers[enemy] or layers[i] != layer_snapshot[i]:
		return
	var direction := movement_snapshot[enemy] - movement_snapshot[i]
	if direction.length_squared() > 0.01:
		headings[i] = lerp_angle(headings[i], direction.angle(), minf(1, dt * 10))
	if direction.length_squared() > 9.0 * 9.0:
		var destination := movement_snapshot[i].move_toward(movement_snapshot[enemy], 46 * dt)
		if colonies[i] == 0 and layers[i] == SURFACE: destination = command_system.safe_step(positions[i], destination)
		if layers[i] == SURFACE or _tunnel_line_clear(positions[i], destination, layers[i]):
			positions[i] = destination
	if roles[i] == DIGGER:
		targets[i] = -1
		paths[i] = []
	else:
		paths[i] = []
		route_goals[i] = -1
	trips[i] = 0

func _resolve_attacks() -> void:
	if not combat_enabled:
		return
	# All movement is complete before either side tests reach or deals damage.
	for i in health.size():
		var enemy := opponents[i]
		if health[i] <= 0 or attack_cooldowns[i] > 0 or layers[i] != layer_snapshot[i]:
			continue
		if enemy < 0 and (roles[i] != INVADER or layers[i] == SURFACE):
			continue
		var bite_bonus := 1.05 if doctrines[colonies[i]] == 2 else 1.0
		if enemy >= 0 and layers[enemy] == layers[i] and positions[i].distance_squared_to(positions[enemy]) <= ATTACK_RANGE * ATTACK_RANGE and (layers[i] == SURFACE or _tunnel_line_clear(positions[i], positions[enemy], layers[i])):
			combat_damage[enemy] += ant_rng[i].randf_range(0.85, 1.15) * bite_bonus
			last_attackers[enemy] = i
			attack_cooldowns[i] = 0.75
		elif enemy < 0 and roles[i] == INVADER and layers[i] >= 0 and layers[i] != colonies[i] and queen_health[layers[i]] > 0:
			var queen := queen_position(layers[i])
			if positions[i].distance_squared_to(queen) <= 18 * 18 and _tunnel_line_clear(positions[i], queen, layers[i]):
				queen_damage[layers[i]] += ant_rng[i].randf_range(0.85, 1.15) * bite_bonus
				attack_cooldowns[i] = 0.75

func _kill_ant(i: int, starved: bool = false) -> void:
	command_system.record_loss(i, starved)
	rival_ai.orders.record_loss(i, starved)
	third_ai.orders.record_loss(i, starved)
	worker_order_ids[i] = -1
	worker_order_kinds[i] = 0
	if selected_ant == i:
		selected_info = ant_info(i)
		selected_info.alive = false
		selected_ant = -1
		following = false
	if not starved and last_attackers[i] >= 0:
		worker_kills[last_attackers[i]] += 1
		if not selected_info.is_empty() and selected_info.id == worker_ids[last_attackers[i]]:
			selected_info.kills = worker_kills[last_attackers[i]]
	if starved:
		starvation_losses[colonies[i]] += 1
	else:
		casualties[colonies[i]] += 1
	remains.append({"pos": positions[i], "colony": colonies[i], "layer": layers[i], "angle": headings[i], "time": 16.0})
	if carrying[i]:
		# A drop can revive an empty patch without changing the food-array size.
		food_index_size = -1
		var dropped := false
		for patch in foods:
			if patch.get("layer", SURFACE) == layers[i] and positions[i].distance_squared_to(patch.pos) < 20 * 20:
				patch.amount += 1
				dropped = true
				break
		if not dropped:
			foods.append({"pos": positions[i], "amount": 1, "initial": 1, "dropped": true, "layer": layers[i]})
	carrying[i] = 0
	targets[i] = -1
	paths[i] = []
	roles[i] = 0
	opponents[i] = -1
	message = "Food stores are empty. Hungry workers are dying." if starved else "A skirmish in the meadow. Fallen carriers drop food that either colony can recover."
	# A surviving queen and fed brood can rebuild even after the last worker dies.

func chamber_cell(colony: int) -> Vector2i:
	if colony == 2: return _cell(THIRD_HOME)
	return HOME_CELL if nest_position(colony) == HOME else Vector2i(COLS - 1 - HOME_CELL.x, HOME_CELL.y)

func entrance_cell(colony: int) -> Vector2i:
	return chamber_cell(colony) + Vector2i(0, -16)

func queen_position(colony: int) -> Vector2:
	return _center(chamber_cell(colony)) + Vector2(-14, -5)

func _nest_grid(colony: int) -> PackedByteArray:
	if colony == 2: return third_earth
	return earth if nest_position(colony) == HOME else rival_earth

func invaders_in(nest: int) -> int:
	if native_enabled and native_kernel: return native_kernel.invaders_in(self, nest)
	var count := 0
	for i in health.size():
		if health[i] > 0 and layers[i] == nest and colonies[i] != nest:
			count += 1
	return count

func launch_raid(colony: int, requested: int = 12) -> int:
	if not invasions_enabled or not match_result.is_empty() or queen_health[colony] <= 0:
		return 0
	var active := 0
	var candidates: Array[int] = []
	for i in health.size():
		if health[i] <= 0 or colonies[i] != colony:
			continue
		if roles[i] == INVADER:
			active += 1
		elif layers[i] == SURFACE and roles[i] == FORAGER and carrying[i] == 0 and opponents[i] < 0:
			candidates.append(i)
	var capacity := mini(command_system.raid_limit() if colony == 0 else 12, maxi(1, living_count(colony) / 4))
	var count := mini(requested, maxi(0, capacity - active))
	var enemy_entrance := nest_position(enemy_for(colony))
	candidates.sort_custom(func(a: int, b: int) -> bool: return positions[a].distance_squared_to(enemy_entrance) < positions[b].distance_squared_to(enemy_entrance))
	var sent := 0
	for i in candidates:
		if sent >= count:
			break
		roles[i] = INVADER
		trips[i] = 0
		targets[i] = -1
		paths[i] = []
		route_goals[i] = -1
		sent += 1
	if sent > 0:
		next_raids[colony] = elapsed + 60
		message = "%s colony sends %d workers to the enemy entrance. R orders your next raid." % ["Your" if colony == 0 else "The rival", sent]
	elif colony == 0:
		message = "No workers ready for a new raid. Keep foraging or wait for the current party."
	_update_ui()
	return sent

func _update_defenses() -> void:
	for colony in colony_count:
		if invaders_in(colony) == 0:
			continue
		var active := 0
		var candidates: Array[int] = []
		for i in health.size():
			if health[i] <= 0 or colonies[i] != colony:
				continue
			if roles[i] == GUARD:
				active += 1
			elif layers[i] == SURFACE and roles[i] == FORAGER and carrying[i] == 0 and opponents[i] < 0:
				candidates.append(i)
		var home := nest_position(colony)
		candidates.sort_custom(func(a: int, b: int) -> bool: return positions[a].distance_squared_to(home) < positions[b].distance_squared_to(home))
		for i in candidates:
			if active >= 6:
				break
			roles[i] = GUARD
			trips[i] = 0
			paths[i] = []
			route_goals[i] = -1
			active += 1

func _recover_worker(i: int, dt: float) -> void:
	if (roles[i] == INVADER or roles[i] == GUARD or roles[i] == WARDEN) and health[i] <= 1.2 and carrying[i] == 0:
		roles[i] = RECOVERING
		worker_order_ids[i] = -1
		worker_order_kinds[i] = 0
		paths[i] = []
		route_goals[i] = -1
	if roles[i] != RECOVERING: return
	var colony: int = colonies[i]
	if layers[i] == SURFACE and positions[i].distance_squared_to(nest_position(colony)) < 30 * 30 and food_stores[colony] > 0:
		var healed := minf(3 - health[i], minf(dt * 0.6, food_stores[colony] * 20))
		health[i] += healed
		food_stores[colony] -= healed / 20
		upkeep_spent[colony] += healed / 20
		if health[i] >= 2.99:
			roles[i] = FORAGER
			worker_order_ids[i] = -1
			worker_order_kinds[i] = 0

func _surface_mission(i: int, dt: float) -> void:
	if roles[i] == RECOVERING:
		_move_surface_worker(i, nest_position(colonies[i]), dt)
		return
	if roles[i] == WARDEN:
		var patrol := worker_order_goals[i] + Vector2.from_angle(i * 2.4 + elapsed * 0.15) * 18
		_move_surface_worker(i, patrol, dt)
		return
	var nest: int = enemy_for(colonies[i]) if roles[i] == INVADER else colonies[i]
	if roles[i] == GUARD and nest_quiet[nest] > 8:
		roles[i] = FORAGER
		return
	var destination := nest_position(nest)
	if roles[i] == INVADER and worker_order_kinds[i] == command_system.RAID:
		var target := worker_order_goals[i]
		for candidate in colony_count:
			if candidate != colonies[i] and queen_health[candidate] > 0 and target.distance_to(nest_position(candidate)) <= 90: nest = candidate; destination = nest_position(nest); break
		if target.distance_to(nest_position(nest)) > 90:
			_move_surface_worker(i, target, dt)
			return
	_move_surface_worker(i, destination, dt)
	if positions[i].distance_squared_to(destination) <= 16 * 16:
		_enter_nest(i, nest)

func _move_surface_worker(i: int, destination: Vector2, dt: float) -> void:
	var direction := destination - positions[i]
	if colonies[i] == 0:
		var detour: Vector2 = command_system.avoidance_direction(positions[i], destination, i)
		if detour.length_squared() > 0.01: direction = detour * maxf(50, direction.length())
	headings[i] = direction.angle()
	var proposed := positions[i].move_toward(positions[i] + direction, 50 * dt)
	positions[i] = command_system.safe_step(positions[i], proposed) if colonies[i] == 0 else proposed

func _enter_nest(i: int, nest: int) -> bool:
	if layers[i] != SURFACE or positions[i].distance_squared_to(nest_position(nest)) > 16 * 16:
		return false
	if _nest_grid(nest)[_idx(entrance_cell(nest))] != 1:
		return false
	layers[i] = nest
	trips[i] = 0
	positions[i] = _center(entrance_cell(nest))
	paths[i] = []
	route_goals[i] = -1
	targets[i] = -1
	if colonies[i] != nest:
		nest_quiet[nest] = 0
		message = "%s nest breached! Defenders are returning through the entrance. Q inspects the tunnels." % ("Your" if nest == 0 else "Rival")
	return true

func _tunnel_mission(i: int, dt: float) -> void:
	var nest := layers[i]
	if roles[i] == INVADER and queen_health[nest] <= 0:
		roles[i] = RETURNING
		paths[i] = []
		route_goals[i] = -1
	if roles[i] == INVADER:
		var queen := queen_position(nest)
		if positions[i].distance_squared_to(queen) > 16 * 16 or not _tunnel_line_clear(positions[i], queen, nest):
			_navigate_tunnel(i, _cell(queen), dt)
		else:
			headings[i] = (queen - positions[i]).angle()
	elif roles[i] == GUARD and nest_quiet[nest] <= 8:
		var target := -1
		var nearest := INF
		for j in health.size():
			if health[j] > 0 and layers[j] == nest and colonies[j] != colonies[i]:
				var distance := positions[i].distance_squared_to(positions[j])
				if distance < nearest:
					nearest = distance
					target = j
		_navigate_tunnel(i, _cell(positions[target]) if target >= 0 else _cell(queen_position(nest)), dt)
	else:
		var recovering: bool = roles[i] == RECOVERING
		if not recovering: roles[i] = RETURNING
		_navigate_tunnel(i, entrance_cell(nest), dt)
		if positions[i].distance_squared_to(_center(entrance_cell(nest))) < 6 * 6:
			layers[i] = SURFACE
			positions[i] = nest_position(nest) + Vector2(0, 18)
			roles[i] = RECOVERING if recovering else FORAGER
			paths[i] = []
			route_goals[i] = -1
			targets[i] = -1

func tunnel_route(start: Vector2i, goal: Vector2i, nest: int) -> Array:
	var grid := _nest_grid(nest)
	if not _valid(start) or not _valid(goal) or grid[_idx(start)] != 1 or grid[_idx(goal)] != 1:
		return []
	var parents := PackedInt32Array()
	parents.resize(CELLS)
	parents.fill(-1)
	var queue: Array[Vector2i] = [start]
	parents[_idx(start)] = _idx(start)
	var head := 0
	while head < queue.size() and parents[_idx(goal)] == -1:
		var current := queue[head]
		head += 1
		for n in _neighbors(current):
			var j := _idx(n)
			if grid[j] == 1 and parents[j] == -1:
				parents[j] = _idx(current)
				queue.append(n)
	if parents[_idx(goal)] == -1:
		return []
	var route: Array = []
	var cursor := _idx(goal)
	while cursor != _idx(start):
		route.push_front(Vector2i(cursor % COLS, cursor / COLS))
		cursor = parents[cursor]
	return route

func _navigate_tunnel(i: int, goal: Vector2i, dt: float) -> void:
	var grid := _nest_grid(layers[i])
	if route_goals[i] != _idx(goal) or (not paths[i].is_empty() and grid[_idx(paths[i][0])] != 1):
		paths[i] = []
		route_goals[i] = _idx(goal)
	if paths[i].is_empty() and _cell(positions[i]) != goal:
		trips[i] -= dt
		if trips[i] > 0:
			return
		paths[i] = tunnel_route(_cell(positions[i]), goal, layers[i])
		if not paths[i].is_empty() and positions[i].distance_squared_to(_center(_cell(positions[i]))) > 2 * 2:
			paths[i].push_front(_cell(positions[i]))
		trips[i] = 0.3
	var destination := _center(goal)
	if not paths[i].is_empty():
		destination = _center(paths[i][0])
	elif _cell(positions[i]) != goal:
		return
	var direction := destination - positions[i]
	if direction.length_squared() > 0.01:
		headings[i] = direction.angle()
	var next := positions[i].move_toward(destination, 47 * dt)
	if _tunnel_line_clear(positions[i], next, layers[i]):
		positions[i] = next
	if not paths[i].is_empty() and positions[i].distance_squared_to(destination) < 0.5:
		paths[i].pop_front()

func _tunnel_line_clear(a: Vector2, b: Vector2, nest: int) -> bool:
	var grid := _nest_grid(nest)
	if not Rect2(Vector2.ZERO, SIZE).has_point(a) or not Rect2(Vector2.ZERO, SIZE).has_point(b):
		return false
	# Traverse every crossed tile, including both sides of an exact grid corner.
	var current := _cell(a)
	var goal := _cell(b)
	var direction := b - a
	var step_cell := Vector2i(int(signf(direction.x)), int(signf(direction.y)))
	var delta_x := TILE / absf(direction.x) if direction.x != 0 else INF
	var delta_y := TILE / absf(direction.y) if direction.y != 0 else INF
	var edge_x := (current.x + (1 if direction.x > 0 else 0)) * TILE
	var edge_y := (current.y + (1 if direction.y > 0 else 0)) * TILE
	var next_x := (edge_x - a.x) / direction.x if direction.x != 0 else INF
	var next_y := (edge_y - a.y) / direction.y if direction.y != 0 else INF
	for crossed in COLS + ROWS + 4:
		if not _valid(current) or grid[_idx(current)] != 1:
			return false
		if current == goal:
			return true
		if is_equal_approx(next_x, next_y):
			var side_x := current + Vector2i(step_cell.x, 0)
			var side_y := current + Vector2i(0, step_cell.y)
			if not _valid(side_x) or not _valid(side_y) or grid[_idx(side_x)] != 1 or grid[_idx(side_y)] != 1:
				return false
			current += step_cell
			next_x += delta_x
			next_y += delta_y
		elif next_x < next_y:
			current.x += step_cell.x
			next_x += delta_x
		else:
			current.y += step_cell.y
			next_y += delta_y
	return false

func _check_outcome() -> void:
	if not match_result.is_empty(): return
	var rivals_alive := 0
	for c in range(1, colony_count):
		if queen_health[c] > 0: rivals_alive += 1
	if queen_health[0] > 0 and rivals_alive > 0: return
	if queen_health[0] <= 0 and rivals_alive == 0:
		match_result = "draw"
		message = "DRAW - All queens died."
	elif rivals_alive == 0:
		match_result = "victory"
		message = "VICTORY - All rival queens have died."
	else:
		match_result = "defeat"
		message = "DEFEAT - Your queen died: %s." % queen_death_causes[0]
	painting = false
	result_dismissed = false
	_update_ui()
	queue_redraw()

func upkeep_rate(colony: int, population: int = -1) -> float:
	if population < 0: population = living_count(colony)
	var rate: float = (0.06 if queen_health[colony] > 0 else 0.0) + population * 0.004 + brood[colony].size() * 0.006
	return rate * (1.15 if doctrines[colony] == 2 else 1.0)

func stage_duration(colony: int, stage: int) -> float:
	return STAGE_TIMES[stage] * (0.85 if doctrines[colony] == 1 else 1.0)

func stage_feed_cost(colony: int, stage: int) -> float:
	# One food to lay; three to feed a larva; three/four to finish pupation.
	return 3.0 if stage == 0 or doctrines[colony] == 1 else 4.0

func brood_counts(colony: int) -> Vector3i:
	var result := Vector3i.ZERO
	for young in brood[colony]:
		result[young.stage] += 1
	return result

func _update_nurseries(dt: float) -> void:
	for colony in colony_count:
		# Count once per colony, then track the deaths/births in this update locally.
		# External callers still get a fresh count from upkeep_rate's default path.
		var population := living_count(colony)
		var required := upkeep_rate(colony, population) * dt
		var paid := minf(food_stores[colony], required)
		food_stores[colony] = maxf(0, food_stores[colony] - paid)
		upkeep_spent[colony] += paid
		var shortage := maxf(0, required - paid) / maxf(required, 0.000001)
		queen_hungry[colony] = 1 if shortage > 0.001 else 0
		if queen_health[colony] > 0:
			if shortage > 0:
				queen_health[colony] = maxf(0, queen_health[colony] - dt * 0.65 * shortage)
				if queen_health[colony] == 0:
						queen_death_causes[colony] = "starvation"
			else:
				queen_health[colony] = minf(100, queen_health[colony] + dt * 0.4)
		if shortage > 0:
			for i in health.size():
				if colonies[i] == colony and health[i] > 0:
					health[i] = maxf(0, health[i] - dt * 0.04 * shortage)
					if health[i] == 0:
						_kill_ant(i, true)
						population -= 1
		# Feeding the colony takes priority. Keep a small reserve before breeding.
		var reserve := maxf(8, upkeep_rate(colony, population) * 15)
		for k in range(brood[colony].size() - 1, -1, -1):
			var young: Dictionary = brood[colony][k]
			if shortage < 0.001 or young.stage == 2:
				young.age += dt
			if young.age < stage_duration(colony, young.stage):
				continue
			if young.stage < 2:
				var cost := stage_feed_cost(colony, young.stage)
				if food_stores[colony] >= cost + reserve:
					food_stores[colony] -= cost
					brood_spent[colony] += cost
					young.stage += 1
					young.age = 0.0
			elif population < POPULATION_CAP:
				_spawn_worker(colony)
				population += 1
				brood[colony].remove_at(k)
		if queen_health[colony] > 0 and shortage < 0.001 and brood[colony].size() < BROOD_CAP and population + brood[colony].size() < POPULATION_CAP:
			var interval := EGG_INTERVAL * (0.85 if doctrines[colony] == 1 else 1.0)
			egg_clocks[colony] = minf(interval, egg_clocks[colony] + dt)
			if egg_clocks[colony] >= interval and food_stores[colony] >= 1 + reserve:
				food_stores[colony] -= 1
				brood_spent[colony] += 1
				brood[colony].append({"stage": 0, "age": 0.0})
				egg_clocks[colony] = 0

func _spawn_worker(colony: int) -> void:
	var slot := -1
	for i in health.size():
		if colonies[i] == colony and health[i] <= 0:
			slot = i
			break
	if slot < 0:
		slot = health.size()
		colony_workers[colony].append(slot)
		var count := slot + 1
		positions.resize(count)
		worker_ids.resize(count)
		born_at.resize(count)
		deliveries.resize(count)
		worker_kills.resize(count)
		walked.resize(count)
		last_attackers.resize(count)
		layers.resize(count)
		route_goals.resize(count)
		headings.resize(count)
		carrying.resize(count)
		roles.resize(count)
		trips.resize(count)
		targets.resize(count)
		colonies.resize(count)
		health.resize(count)
		attack_cooldowns.resize(count)
		worker_order_ids.resize(count)
		worker_order_kinds.resize(count)
		worker_order_goals.resize(count)
		opponents.resize(count)
		hit_flashes.resize(count)
		combat_damage.resize(count)
		paths.append([])
		ant_rng.append(RandomNumberGenerator.new())
	colonies[slot] = colony
	worker_order_ids[slot] = -1
	worker_order_kinds[slot] = 0
	worker_order_goals[slot] = Vector2.ZERO
	worker_ids[slot] = next_worker_id
	next_worker_id += 1
	born_at[slot] = elapsed
	deliveries[slot] = 0
	worker_kills[slot] = 0
	walked[slot] = 0
	last_attackers[slot] = -1
	layers[slot] = SURFACE
	route_goals[slot] = -1
	ant_rng[slot].seed = world_seed + slot * 104729 + (births[colony] + 1) * 1000003
	var angle := ant_rng[slot].randf() * TAU
	if swapped_sides: angle = PI - angle
	positions[slot] = nest_position(colony) + Vector2.from_angle(angle) * 15
	headings[slot] = angle
	health[slot] = 3
	carrying[slot] = 0
	roles[slot] = 0
	trips[slot] = 0
	targets[slot] = -1
	opponents[slot] = -1
	hit_flashes[slot] = 0
	combat_damage[slot] = 0
	attack_cooldowns[slot] = 0.75
	paths[slot] = []
	for i in opponents.size():
		if opponents[i] == slot:
			opponents[i] = -1
	births[colony] += 1

func _forage(i: int, dt: float) -> void:
	var p := positions[i]
	var worker_rng := ant_rng[i]
	var nest := nest_for(i)
	var own_food_scent: PackedFloat32Array = [food_scent, rival_food_scent, third_food_scent][colonies[i]]
	var own_home_scent: PackedFloat32Array = [home_scent, rival_home_scent, third_home_scent][colonies[i]]
	var angle := headings[i]
	trips[i] += dt
	var desired := Vector2.from_angle(angle)
	var travel_goal := p + desired * 150
	if carrying[i]:
		travel_goal = nest
		desired = (nest - p).normalized()
		angle = lerp_angle(angle, desired.angle(), minf(1, dt * 6))
		_deposit(own_food_scent, p, 0.6 / (1.0 + trips[i] * 0.018))
		if p.distance_squared_to(nest) < 23 * 23:
			carrying[i] = 0
			if colonies[i] == 0:
				delivered += 1
			elif colonies[i] == 1:
				rival_delivered += 1
			else: third_delivered += 1
			food_stores[colonies[i]] += 1
			deliveries[i] += 1
			trips[i] = 0
			angle += PI + worker_rng.randf_range(-0.6, 0.6) * (-1 if swapped_sides else 1)
			if colonies[i] == 0 and delivered == 1:
				message = "First food delivered. Returning workers leave a trail for other scouts."
	else:
		_deposit(own_home_scent, p, 0.16 / (1.0 + trips[i] * 0.06))
		var nearest := -1
		var best_dist := 86.0 * 86.0
		nearest = _nearest_food(p, colonies[i] == 0)
		if nearest >= 0: best_dist = p.distance_squared_to(foods[nearest].pos)
		if nearest >= 0:
			var food_pos: Vector2 = foods[nearest].pos
			travel_goal = food_pos
			angle = lerp_angle(angle, (food_pos - p).angle(), minf(1, dt * 7))
			if best_dist < 15 * 15:
				foods[nearest].amount -= 1
				carrying[i] = 1
				trips[i] = 0
				angle = (nest - p).angle()
		else:
			# Three local antenna samples. Keep heading inertia to avoid orbiting a peak.
			var best_angle := angle
			var forward := _sample(own_food_scent, p + Vector2.from_angle(angle) * 27) * 1.06
			var left := _sample(own_food_scent, p + Vector2.from_angle(angle - 0.65) * 27)
			var right := _sample(own_food_scent, p + Vector2.from_angle(angle + 0.65) * 27)
			# Compare both antennae together; sequential thresholding favored one turn.
			best_angle += scent_turn(forward, left, right)
			angle = lerp_angle(angle, best_angle, dt * 3.5)
			angle += worker_rng.randf_range(-1, 1) * 1.9 * sqrt(dt) * (-1 if swapped_sides else 1)
			if p.distance_squared_to(nest) < 38 * 38:
				angle = lerp_angle(angle, (p - nest).angle(), dt * 3)
			travel_goal = p + Vector2.from_angle(angle) * 150
			if worker_order_kinds[i] == command_system.FORAGE and p.distance_squared_to(worker_order_goals[i]) > 90 * 90:
				travel_goal = worker_order_goals[i]
				angle = lerp_angle(angle, (travel_goal - p).angle(), minf(1, dt * 4))
	if colonies[i] == 0 and not avoid_points.is_empty():
		var detour: Vector2 = command_system.avoidance_direction(p, travel_goal, i)
		if detour.length_squared() > 0.01: angle = lerp_angle(angle, detour.angle(), minf(1, dt * 10))
	if food_stores[colonies[i]] > 0 and p.distance_squared_to(nest) < 30 * 30:
		health[i] = minf(3, health[i] + dt * 0.6)
	var forage_bonus := 1.08 if doctrines[colonies[i]] == 1 else 1.0
	var velocity := Vector2.from_angle(angle) * (43.0 if carrying[i] else 52.0) * forage_bonus
	var proposed := p + velocity * dt
	p = command_system.safe_step(p, proposed) if colonies[i] == 0 else proposed
	if p.x < 9 or p.x > SIZE.x - 9:
		angle = PI - angle
		p.x = clampf(p.x, 9, SIZE.x - 9)
	if p.y < 9 or p.y > SIZE.y - 9:
		angle = -angle
		p.y = clampf(p.y, 9, SIZE.y - 9)
	positions[i] = p
	headings[i] = angle

func _rebuild_food_index() -> void:
	food_buckets.clear()
	food_neighbors.clear()
	for f in foods.size():
		var patch: Dictionary = foods[f]
		if patch.amount <= 0 or patch.get("layer", SURFACE) != SURFACE: continue
		var bucket := Vector2i(Vector2(patch.pos) / FOOD_BUCKET_SIZE)
		if not food_buckets.has(bucket): food_buckets[bucket] = []
		food_buckets[bucket].append(f)
	food_index_size = foods.size()

func _nearest_food(p: Vector2, respect_avoid: bool = false) -> int:
	if food_index_size != foods.size(): _rebuild_food_index()
	var cell := Vector2i(p / FOOD_BUCKET_SIZE)
	if not food_neighbors.has(cell):
		var nearby: Array = []
		for y in range(-1, 2):
			for x in range(-1, 2):
				var bucket = food_buckets.get(cell + Vector2i(x, y))
				if bucket != null: nearby.append_array(bucket)
		food_neighbors[cell] = nearby
	var nearest := -1
	var best_dist := 86.0 * 86.0
	for f in food_neighbors[cell]:
		if foods[f].amount <= 0: continue
		if respect_avoid and command_system.is_avoided(foods[f].pos): continue
		var distance: float = p.distance_squared_to(foods[f].pos)
		if distance < best_dist or (nearest >= 0 and distance == best_dist and f < nearest):
			best_dist = distance
			nearest = f
	return nearest

func _deposit(field: PackedFloat32Array, p: Vector2, strength: float) -> void:
	var c := _cell(p)
	var j := _idx(c)
	field[j] = minf(5.0, field[j] + strength * STEP * 9)
	if scent_active[j] == 0:
		scent_active[j] = 1
		active_scent.append(j)
	trails_dirty = true
	# Keep the same four deposits without allocating neighbor arrays or vectors.
	var spread := strength * STEP * 2.2
	if c.x > 0:
		field[j - 1] = minf(5.0, field[j - 1] + spread)
		if scent_active[j - 1] == 0:
			scent_active[j - 1] = 1
			active_scent.append(j - 1)
	if c.x < COLS - 1:
		field[j + 1] = minf(5.0, field[j + 1] + spread)
		if scent_active[j + 1] == 0:
			scent_active[j + 1] = 1
			active_scent.append(j + 1)
	if c.y > 0:
		field[j - COLS] = minf(5.0, field[j - COLS] + spread)
		if scent_active[j - COLS] == 0:
			scent_active[j - COLS] = 1
			active_scent.append(j - COLS)
	if c.y < ROWS - 1:
		field[j + COLS] = minf(5.0, field[j + COLS] + spread)
		if scent_active[j + COLS] == 0:
			scent_active[j + COLS] = 1
			active_scent.append(j + COLS)

func _sample(field: PackedFloat32Array, p: Vector2) -> float:
	return field[clampi(int(p.y / TILE), 0, ROWS - 1) * COLS + clampi(int(p.x / TILE), 0, COLS - 1)]

func scent_turn(forward: float, left: float, right: float) -> float:
	if maxf(left, right) > forward + 0.025 and absf(left - right) > 0.025:
		return -0.65 if left > right else 0.65
	return 0.0

func pending_count() -> int:
	return designated_cells.size()

func digger_count() -> int:
	var result := 0
	for i in health.size():
		if colonies[i] == 0 and health[i] > 0:
			result += 1 if roles[i] == DIGGER else 0
	return result

func _allocate_workers() -> void:
	var needed := mini(command_system.digging_limit(), living_count(0)) if pending_count() > 0 else 0
	var active := digger_count()
	if active < needed:
		# Recruit the nearest empty-handed ants through the shared nest entrance.
		var available: Array[int] = []
		for i in health.size():
			if colonies[i] == 0 and health[i] > 0 and roles[i] == FORAGER and layers[i] == SURFACE and carrying[i] == 0 and opponents[i] == -1:
				available.append(i)
		available.sort_custom(func(a: int, b: int) -> bool: return positions[a].distance_squared_to(HOME) < positions[b].distance_squared_to(HOME))
		for i in available:
			if active >= needed:
				break
			roles[i] = 1
			trips[i] = 0
			targets[i] = -1
			paths[i] = []
			active += 1
	elif active > needed:
		for i in health.size():
			if active <= needed: break
			if colonies[i] == 0 and roles[i] == 1:
				roles[i] = RETURNING if layers[i] >= 0 else FORAGER
				trips[i] = 0
				targets[i] = -1
				paths[i] = []
				active -= 1
		message = "Excavation complete. Workers return to foraging." if needed == 0 and pending_count() == 0 else "Digging priority changed; unfinished orders remain queued."

func _dig(i: int, dt: float) -> void:
	var target := targets[i]
	if target >= 0 and (orders[target] == 0 or earth[target] != 0):
		targets[i] = -1
		paths[i] = []
		target = -1
	if target < 0:
		trips[i] -= dt
		if trips[i] > 0:
			return
		_assign_dig_job(i)
		trips[i] = rng.randf_range(0.3, 0.6)
		return
	var path: Array = paths[i]
	if not path.is_empty():
		var destination := _center(path[0]) + Vector2.from_angle(i * 2.4) * 2.8
		var direction := destination - positions[i]
		headings[i] = direction.angle()
		positions[i] = positions[i].move_toward(destination, dt * 47)
		if positions[i].distance_squared_to(destination) < 1:
			path.pop_front()
		return
	var target_center := _center(Vector2i(target % COLS, target / COLS))
	headings[i] = (target_center - positions[i]).angle()
	excavation[target] += dt * 0.34
	if excavation[target] >= 1.0:
		earth[target] = 1
		_set_order(target, false)
		open_cells[0][target] = true
		terrain_images[0].set_pixel(target % COLS, target / COLS, Color("1c201b"))
		terrain_dirty[0] = 1
		dug += 1
		targets[i] = -1
		trips[i] = 0

func _assign_dig_job(i: int) -> void:
	# One BFS through reachable tunnels finds every accessible work face.
	var start := _cell(positions[i])
	var parents := PackedInt32Array()
	parents.resize(CELLS)
	parents.fill(-1)
	var queue: Array[Vector2i] = [start]
	parents[_idx(start)] = _idx(start)
	var head := 0
	var best := -1
	var stand := start
	var best_score := INF
	var assigned_counts := {}
	for target in targets:
		if target >= 0: assigned_counts[target] = assigned_counts.get(target, 0) + 1
	while head < queue.size():
		var c := queue[head]
		head += 1
		for n in _neighbors(c):
			var j := _idx(n)
			if earth[j] == 0 and orders[j] == 1:
				var assigned: int = assigned_counts.get(j, 0)
				# Only two ants fit at one work face; the rest wait for room.
				if assigned >= 2:
					continue
				var score := positions[i].distance_squared_to(_center(c)) + assigned * 22000.0
				if score < best_score:
					best_score = score
					best = j
					stand = c
			if earth[j] == 1 and parents[j] == -1:
				parents[j] = _idx(c)
				queue.append(n)
	if best < 0:
		return
	targets[i] = best
	var route: Array = []
	var cursor := _idx(stand)
	while cursor != _idx(start):
		route.push_front(Vector2i(cursor % COLS, cursor / COLS))
		cursor = parents[cursor]
	paths[i] = route

func designate(c: Vector2i, connect_to_nest: bool = true, erase: bool = false) -> void:
	if not _valid(c) or not match_result.is_empty():
		return
	if connect_to_nest and not erase and earth[_idx(c)] == 0:
		_connect_designation(c)
	for y in range(c.y - 1, c.y + 2):
		for x in range(c.x - 1, c.x + 2):
			var n := Vector2i(x, y)
			if _valid(n) and earth[_idx(n)] == 0:
				_set_order(_idx(n), not erase)
	message = "Designation erased. Excavated tunnels remain open." if erase else "Tunnel marked. Up to 12 workers will excavate from the nearest open tunnel."

func _connect_designation(start: Vector2i) -> void:
	# Search through soil, routing around stone, to connect a new stroke to the nest.
	var parents := PackedInt32Array()
	parents.resize(CELLS)
	parents.fill(-1)
	var queue: Array[Vector2i] = [start]
	parents[_idx(start)] = _idx(start)
	var head := 0
	var goal := -1
	while head < queue.size():
		var c := queue[head]
		head += 1
		var j := _idx(c)
		if earth[j] == 1:
			goal = j
			break
		for n in _neighbors(c):
			var k := _idx(n)
			if earth[k] != 2 and parents[k] == -1:
				parents[k] = j
				queue.append(n)
	if goal == -1:
		return
	var cursor := goal
	while cursor != _idx(start):
		if earth[cursor] == 0:
			_set_order(cursor, true)
		cursor = parents[cursor]
	_set_order(_idx(start), true)

func _rebuild_map_images() -> void:
	terrain_images.clear()
	terrain_textures.clear()
	open_cells = [{}, {}, {}]
	for side in colony_count:
		var grid := earth if side == 0 else (rival_earth if side == 1 else third_earth)
		var picture := Image.create(COLS, ROWS, false, Image.FORMAT_RGBA8)
		for y in ROWS:
			for x in COLS:
				var j := y * COLS + x
				var color := Color("44372c") if (x * 7 + y * 11) % 13 else Color("493b2e")
				if grid[j] == 1:
					color = Color("1c201b")
					open_cells[side][j] = true
				elif grid[j] == 2: color = Color("626059")
				picture.set_pixel(x, y, color)
		terrain_images.append(picture)
		terrain_textures.append(ImageTexture.create_from_image(picture))
	terrain_dirty.fill(0)
	trail_image = Image.create(COLS, ROWS, false, Image.FORMAT_RGBA8)
	trail_image.fill(Color.TRANSPARENT)
	trail_texture = ImageTexture.create_from_image(trail_image)
	trails_dirty = true

func _refresh_map_images() -> void:
	for side in colony_count:
		if terrain_dirty[side]:
			terrain_textures[side].update(terrain_images[side])
			terrain_dirty[side] = 0
	if trails_dirty and not underground:
		if native_enabled and native_kernel:
			trail_image.set_data(COLS, ROWS, false, Image.FORMAT_RGBA8, native_kernel.trail_pixels(self))
		else:
			for j in active_scent:
				var color := Color(0.38, 0.7, 0.85, minf(home_scent[j] * 0.065, 0.16))
				color = color.blend(Color(0.68, 0.88, 0.4, minf(food_scent[j] * 0.115, 0.48)))
				color = color.blend(Color(0.96, 0.43, 0.46, minf(rival_food_scent[j] * 0.115, 0.48)))
				if colony_count == 3: color = color.blend(Color(0.35, 0.68, 0.96, minf(third_food_scent[j] * 0.115, 0.48)))
				trail_image.set_pixel(j % COLS, j / COLS, color)
		trail_texture.update(trail_image)
		trails_dirty = false

func screen_to_world(point: Vector2) -> Vector2:
	return camera_center + (point - ORIGIN - VIEW_SIZE * 0.5) / camera_zoom

func world_to_screen(point: Vector2) -> Vector2:
	return ORIGIN + VIEW_SIZE * 0.5 + (point - camera_center) * camera_zoom

func visible_world_rect() -> Rect2:
	return Rect2(camera_center - VIEW_SIZE / camera_zoom * 0.5, VIEW_SIZE / camera_zoom).intersection(Rect2(Vector2.ZERO, SIZE))

func _sync_camera() -> void:
	camera_zoom = clampf(camera_zoom, 0.33333334, 3.5)
	var half := VIEW_SIZE / camera_zoom * 0.5
	camera_center.x = clampf(camera_center.x, minf(half.x, SIZE.x * 0.5), maxf(SIZE.x - half.x, SIZE.x * 0.5))
	camera_center.y = clampf(camera_center.y, minf(half.y, SIZE.y * 0.5), maxf(SIZE.y - half.y, SIZE.y * 0.5))
	world_canvas.scale = Vector2.ONE * camera_zoom
	world_canvas.position = VIEW_SIZE * 0.5 - camera_center * camera_zoom
	decor_canvas.scale = world_canvas.scale
	decor_canvas.position = world_canvas.position
	decor_canvas.visible = not underground
	world_canvas.queue_redraw()
	mini_canvas.queue_redraw()

func _activate_camera_view() -> void:
	var next := inspected_colony if underground else SURFACE
	if next != active_camera_layer:
		camera_centers[active_camera_layer + 1] = camera_center
		camera_zooms[active_camera_layer + 1] = camera_zoom
		active_camera_layer = next
		camera_center = camera_centers[next + 1]
		camera_zoom = camera_zooms[next + 1]
		dragging = false
		painting = false
	_sync_camera()

func zoom_at(point: Vector2, factor: float) -> void:
	var anchor := screen_to_world(point)
	camera_zoom = clampf(camera_zoom * factor, 0.33333334, 3.5)
	camera_center = anchor - (point - ORIGIN - VIEW_SIZE * 0.5) / camera_zoom
	following = false
	_sync_camera()

func focus_nest() -> void:
	following = false
	camera_center = _center(chamber_cell(inspected_colony)) if underground else nest_position(0)
	camera_zoom = 1.3 if underground else 1.0
	_sync_camera()
	_update_ui()

func fit_world() -> void:
	following = false
	camera_center = SIZE * 0.5
	camera_zoom = minf(VIEW_SIZE.x / SIZE.x, VIEW_SIZE.y / SIZE.y)
	_sync_camera()
	_update_ui()

func _toggle_dig_mode() -> void:
	if not match_result.is_empty(): return
	if not underground or inspected_colony != 0:
		inspected_colony = 0
		set_view(true)
		dig_mode = true
	else: dig_mode = not dig_mode
	painting = false
	_update_ui()

func _map_ui_contains(point: Vector2) -> bool:

func _pick_ant(point: Vector2) -> int:
	var best := -1
	var distance := pow(maxf(7, 10 / camera_zoom), 2)
	for i in health.size():
		if health[i] > 0 and layers[i] == active_camera_layer:
			var d := positions[i].distance_squared_to(point)
			if d < distance:
				distance = d
				best = i
	return best

func select_ant(i: int) -> void:
	selected_ant = i
	following = false
	selected_info = ant_info(i) if i >= 0 else {}
	_update_ui()

func ant_info(i: int) -> Dictionary:
	var moving_speed := 47.0 if layers[i] >= 0 else 50.0
	if opponents[i] >= 0: moving_speed = 46.0
	elif roles[i] == FORAGER:
		moving_speed = (43.0 if carrying[i] else 52.0) * (1.08 if doctrines[colonies[i]] == 1 else 1.0)
	return {"id": worker_ids[i], "colony": colonies[i], "alive": health[i] > 0, "health": health[i], "age": maxf(0, elapsed - born_at[i]), "role": roles[i], "layer": layers[i], "cargo": carrying[i], "delivered": deliveries[i], "kills": worker_kills[i], "walked": walked[i], "speed": moving_speed, "bite": 1.05 if doctrines[colonies[i]] == 2 else 1.0, "cooldown": attack_cooldowns[i], "pos": positions[i], "order": "Recover at own nest" if roles[i] == RECOVERING else ("%s #%d" % [command_system.NAMES[worker_order_kinds[i]], worker_order_ids[i]] if worker_order_ids[i] >= 0 else "Autonomous")}

func _input(event: InputEvent) -> void:
	if event is InputEventKey and event.pressed and not event.echo:
		match event.keycode:
			KEY_TAB:
				set_view(not underground)
				get_viewport().set_input_as_handled()
			KEY_SPACE: _toggle_pause()
			KEY_F: _toggle_trails()
			KEY_Q: _inspect_nursery()
			KEY_I: toggle_rival_plan()
			KEY_R: place_surface_order(command_system.RAID, nest_position(enemy_for(0)))
			KEY_4: set_command_tool(1)
			KEY_5: set_command_tool(2)
			KEY_6: set_command_tool(3)
			KEY_7: set_command_tool(4)
			KEY_L: toggle_labor_panel()
			KEY_G: _toggle_dig_mode()
			KEY_H, KEY_HOME: focus_nest()
			KEY_0: fit_world()
			KEY_C: _toggle_follow()
			KEY_ESCAPE:
				command_tool = 0
				labor_panel.visible = false
				rival_panel.visible = false
				select_ant(-1)
				dig_mode = false
				painting = false
				dragging = false
			KEY_BRACKETLEFT: _set_speed(0.25)
			KEY_BRACKETRIGHT: _set_speed(0.5)
			KEY_1: _set_speed(1)
			KEY_2: _set_speed(2)
			KEY_3: _set_speed(4)
	if event is InputEventMouseButton:
		var point: Vector2 = get_global_transform_with_canvas().affine_inverse() * event.position
		if not event.pressed:
			if painting:
				painting = false
				last_painted = Vector2i(-1, -1)
				get_viewport().set_input_as_handled()
			if dragging and event.button_index == drag_button:
				if not drag_moved and drag_button == MOUSE_BUTTON_LEFT and MAP_RECT.has_point(point) and not _map_ui_contains(point):
					var world_point := screen_to_world(point)
					if command_tool > 0 and not underground:
						place_surface_order(command_tool, world_point)
						dragging = false
						get_viewport().set_input_as_handled()
						return
					var picked := _pick_ant(world_point)
					if picked >= 0:
						select_ant(picked)
					else:
						select_ant(-1)
						if not underground:
							for colony in colony_count:
								if world_point.distance_squared_to(nest_position(colony)) < 30 * 30:
									inspected_colony = colony
									set_view(true)
									break
				if not drag_moved and drag_button == MOUSE_BUTTON_RIGHT and not underground and MAP_RECT.has_point(point) and not _map_ui_contains(point):
					if not command_system.remove_at(screen_to_world(point), maxf(22, 20 / camera_zoom)): command_tool = 0
					_update_ui()
				dragging = false
				get_viewport().set_input_as_handled()
			return
		if MINI_RECT.has_point(point):
			if event.button_index == MOUSE_BUTTON_LEFT and MINI_MAP.has_point(point - MINI_RECT.position):
				camera_center = (point - MINI_RECT.position - MINI_MAP.position) / MINI_MAP.size * SIZE
				following = false
				_sync_camera()
			get_viewport().set_input_as_handled()
			return
		if not MAP_RECT.has_point(point) or _map_ui_contains(point): return
		if event.button_index == MOUSE_BUTTON_WHEEL_UP or event.button_index == MOUSE_BUTTON_WHEEL_DOWN:
			zoom_at(point, 1.15 if event.button_index == MOUSE_BUTTON_WHEEL_UP else 1.0 / 1.15)
			_update_ui()
			get_viewport().set_input_as_handled()
			return
		if dig_mode and underground and inspected_colony == 0 and match_result.is_empty() and event.button_index in [MOUSE_BUTTON_LEFT, MOUSE_BUTTON_RIGHT]:
			painting = true
			erasing = event.button_index == MOUSE_BUTTON_RIGHT
			last_painted = _cell(screen_to_world(point))
			designate(last_painted, true, erasing)
			get_viewport().set_input_as_handled()
		elif event.button_index in [MOUSE_BUTTON_LEFT, MOUSE_BUTTON_RIGHT, MOUSE_BUTTON_MIDDLE]:
			dragging = true
			following = false
			drag_moved = false
			drag_start = point
			drag_last = point
			drag_button = event.button_index
			get_viewport().set_input_as_handled()
	if event is InputEventMouseMotion:
		var point: Vector2 = get_global_transform_with_canvas().affine_inverse() * event.position
		if dragging:
			if point.distance_to(drag_start) > 5: drag_moved = true
			if drag_moved:
				camera_center -= (point - drag_last) / camera_zoom
				following = false
				_sync_camera()
			drag_last = point
			get_viewport().set_input_as_handled()
		elif painting and underground and inspected_colony == 0 and MAP_RECT.has_point(point) and not _map_ui_contains(point):
			var world_point := screen_to_world(point)
			if not Rect2(Vector2.ZERO, SIZE).has_point(world_point): return
			var c := _cell(world_point)
			if c != last_painted:
				var distance := maxi(absi(c.x - last_painted.x), absi(c.y - last_painted.y))
				for s in range(1, distance + 1):
					var v := Vector2(last_painted).lerp(Vector2(c), float(s) / distance)
					designate(Vector2i(roundi(v.x), roundi(v.y)), false, erasing)
				last_painted = c
			get_viewport().set_input_as_handled()
func set_view(value: bool) -> void:

	if value: command_tool = 0
	painting = false
	dragging = false
	if not value or inspected_colony != 0: dig_mode = false
	following = false
	_activate_camera_view()
	_update_ui()
	queue_redraw()

func _idx(c: Vector2i) -> int:
	return c.y * COLS + c.x

func _cell(p: Vector2) -> Vector2i:
	return Vector2i(clampi(int(p.x / TILE), 0, COLS - 1), clampi(int(p.y / TILE), 0, ROWS - 1))

func _center(c: Vector2i) -> Vector2:
	return (Vector2(c) + Vector2(0.5, 0.5)) * TILE

func _valid(c: Vector2i) -> bool:
	return c.x >= 0 and c.x < COLS and c.y >= 0 and c.y < ROWS

func _neighbors(c: Vector2i) -> Array[Vector2i]:
	var result: Array[Vector2i] = []
	for offset in [Vector2i.LEFT, Vector2i.RIGHT, Vector2i.UP, Vector2i.DOWN]:
		var n: Vector2i = c + offset
		if _valid(n):
			result.append(n)
	return result

func _draw_world(canvas: CanvasItem) -> void:
	canvas.draw_set_transform(Vector2.ZERO)
	if underground:
		_draw_underground(canvas)
	else:
		_draw_surface(canvas)
	visible_ant_count = 0
	var visible := visible_world_rect().grow(18)
	batch_counts.fill(0)
	_resize_ant_batches()
	if native_enabled and native_kernel:
		var batch: Dictionary = native_kernel.visual_batch(self, visible, active_camera_layer, camera_zoom)
		batch_counts = batch.counts
		visible_ant_count = batch.visible
		for b in 3: ant_buffers[b] = batch.buffers[b]
	else:
		for i in health.size():
			if health[i] > 0 and layers[i] == active_camera_layer and visible.has_point(positions[i]):
				_batch_ant(i)
				visible_ant_count += 1
	for b in 3:
		ant_batches[b].buffer = ant_buffers[b]
		ant_batches[b].visible_instance_count = batch_counts[b]
		if batch_counts[b] > 0: canvas.draw_multimesh(ant_batches[b], null)
	if camera_zoom >= 0.65:
		for i in health.size():
			if health[i] > 0 and layers[i] == active_camera_layer and visible.has_point(positions[i]) and (opponents[i] >= 0 or roles[i] == INVADER or roles[i] == GUARD or roles[i] == WARDEN or health[i] < 2.95):
				_draw_ant_status(canvas, i)
	canvas.draw_set_transform(Vector2.ZERO)
	_draw_command_overlay(canvas)
	if selected_ant >= 0 and health[selected_ant] > 0 and layers[selected_ant] == active_camera_layer:
		canvas.draw_arc(positions[selected_ant], 11, 0, TAU, 28, Color.WHITE, 1.5 / camera_zoom)
	if dig_mode and underground and inspected_colony == 0 and match_result.is_empty():
		var pointer := get_local_mouse_position()
		if MAP_RECT.has_point(pointer) and not _map_ui_contains(pointer):
			var c := _cell(screen_to_world(pointer))
			var rect := Rect2(Vector2(c - Vector2i.ONE) * TILE, Vector2.ONE * TILE * 3).intersection(Rect2(Vector2.ZERO, SIZE))
			canvas.draw_rect(rect, Color(0.95, 0.76, 0.42, 0.10))
			canvas.draw_rect(rect, ACCENT, false, 1.5 / camera_zoom)
func _draw_command_overlay(canvas: CanvasItem) -> void:
	if not underground:
		var shown: Array = command_system.markers
		if is_instance_valid(rival_panel) and rival_panel.visible:
			shown = shown.duplicate()
			for order in (rival_ai.orders.markers + third_ai.orders.markers if colony_count == 3 else rival_ai.orders.markers):
				var copy: Dictionary = order.duplicate()
				copy["rival"] = true
				copy["colony"] = 2 if order.id >= 500000 else 1
				shown.append(copy)
		for marker in shown:
			var p: Vector2 = marker.pos
			var tint: Color = colony_color(marker.get("colony", 1)) if marker.get("rival", false) else command_system.COLORS[marker.kind]
			var radius := 100.0 if marker.kind == command_system.AVOID else (90.0 if marker.kind == command_system.GUARD else 24.0)
			if not visible_world_rect().grow(radius + 70 / camera_zoom).has_point(p): continue
			var fill := tint
			fill.a = 0.07
			canvas.draw_circle(p, radius, fill)
			canvas.draw_arc(p, radius, 0, TAU, 48, tint, 1.4 / camera_zoom)
			canvas.draw_line(p + Vector2(-9, 0) / camera_zoom, p + Vector2(9, 0) / camera_zoom, tint, 1.5 / camera_zoom)
			canvas.draw_line(p + Vector2(0, -9) / camera_zoom, p + Vector2(0, 9) / camera_zoom, tint, 1.5 / camera_zoom)
			var title: String = command_system.NAMES[marker.kind].to_upper()
			if marker.get("rival", false): title = COLONY_NAMES[marker.get("colony", 1)].to_upper() + " " + title
			if marker.kind != command_system.AVOID: title += " %d / %s" % [marker.assigned, marker.status]
			# Labels retain screen size at overview zoom; workers stay batched below.
			canvas.draw_set_transform(p + Vector2(0, -radius), 0, Vector2.ONE / camera_zoom)
			var width := regular_font.get_string_size(title, HORIZONTAL_ALIGNMENT_LEFT, -1, 11).x
			canvas.draw_rect(Rect2(-width / 2 - 5, -21, width + 10, 18), Color("14251c"))
			_draw_text(canvas, Vector2(-width / 2, -8), title, 11, tint)
			canvas.draw_set_transform(Vector2.ZERO)
	if show_loss_hotspot and not command_system.hotspot.is_empty() and command_system.hotspot.layer == active_camera_layer:
		var spot: Dictionary = command_system.hotspot
		canvas.draw_arc(spot.pos, 45, 0, TAU, 40, RIVAL, 2 / camera_zoom)
		canvas.draw_set_transform(spot.pos, 0, Vector2.ONE / camera_zoom)
		_draw_text(canvas, Vector2(-65, -74), "LOSS HOTSPOT / %d" % spot.count, 12, RIVAL)
		canvas.draw_set_transform(Vector2.ZERO)

func _draw_underground(canvas: CanvasItem) -> void:
	var side := inspected_colony if inspected_colony == 2 else (0 if nest_position(inspected_colony) == HOME else 1)
	var visible_earth := _nest_grid(inspected_colony)
	var visible := visible_world_rect().grow(TILE)

func _mesh_line(vertices: PackedVector3Array, a: Vector2, b: Vector2, width: float) -> void:
	var n := (b - a).orthogonal().normalized() * width * 0.5
	for p in [a - n, a + n, b + n, a - n, b + n, b - n]:
		vertices.append(Vector3(p.x, p.y, 0))

func _mesh_circle(vertices: PackedVector3Array, center: Vector2, radius: float) -> void:
	for k in 16:
		for p in [center, center + Vector2.from_angle(k * TAU / 16) * radius, center + Vector2.from_angle((k + 1) * TAU / 16) * radius]:
			vertices.append(Vector3(p.x, p.y, 0))

func _build_ant_batches() -> void:
	# Reusable meshes contain only appearance; simulation arrays still own every ant.
	for b in 3:
		var vertices := PackedVector3Array()
		if b == 0:
			for side in [-1, 1]:
				for k in 3:
					var x := float(k - 1) * 2.2
					_mesh_line(vertices, Vector2(x, side), Vector2(x - 1.8, side * 3.8), 0.85)
			_mesh_circle(vertices, Vector2(-3, 0), 2.5)
			_mesh_circle(vertices, Vector2.ZERO, 1.6)
			_mesh_circle(vertices, Vector2(2.8, 0), 1.9)
			_mesh_line(vertices, Vector2(3.5, -0.8), Vector2(6, -2.3), 0.8)
			_mesh_line(vertices, Vector2(3.5, 0.8), Vector2(6, 2.3), 0.8)
		elif b == 1:
			_mesh_line(vertices, Vector2(-3, 0), Vector2(3, 0), 3)
		else:
			_mesh_circle(vertices, Vector2(6.5, 0), 2.8)
		var arrays := []
		arrays.resize(Mesh.ARRAY_MAX)
		arrays[Mesh.ARRAY_VERTEX] = vertices
		var mesh := ArrayMesh.new()
		mesh.add_surface_from_arrays(Mesh.PRIMITIVE_TRIANGLES, arrays)
		var batch := MultiMesh.new()
		batch.transform_format = MultiMesh.TRANSFORM_2D
		batch.use_colors = true
		batch.mesh = mesh
		ant_batches.append(batch)
		ant_buffers.append(PackedFloat32Array())

func _resize_ant_batches() -> void:
	for b in 3:
		if ant_batches[b].instance_count != health.size():
			ant_batches[b].instance_count = health.size()
			ant_buffers[b].resize(health.size() * 12)

func _batch_instance(b: int, p: Vector2, angle: float, color: Color) -> void:
	var j := batch_counts[b] * 12
	# MultiMesh's 2D buffer stores two padded transform rows then RGBA.
	var c := cos(angle)
	var s := sin(angle)
	ant_buffers[b][j] = c
	ant_buffers[b][j + 1] = -s
	ant_buffers[b][j + 2] = 0
	ant_buffers[b][j + 3] = p.x
	ant_buffers[b][j + 4] = s
	ant_buffers[b][j + 5] = c
	ant_buffers[b][j + 6] = 0
	ant_buffers[b][j + 7] = p.y
	ant_buffers[b][j + 8] = color.r
	ant_buffers[b][j + 9] = color.g
	ant_buffers[b][j + 10] = color.b
	ant_buffers[b][j + 11] = color.a
	batch_counts[b] += 1

func _batch_ant(i: int) -> void:
	var color := Color("f0dcab") if roles[i] == FORAGER else ACCENT
	if colonies[i] > 0: color = colony_color(colonies[i])
	if hit_flashes[i] > 0: color = Color.WHITE
	_batch_instance(1 if camera_zoom < 0.65 else 0, positions[i], headings[i], color)
	if carrying[i]: _batch_instance(2, positions[i], headings[i], GREEN)
